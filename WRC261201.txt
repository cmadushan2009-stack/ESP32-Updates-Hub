// ------------------- SAFE HARDWARE HELPERS -------------------
void stopMotors() {
  ledcWrite(IN1_PIN, 0);
  ledcWrite(IN2_PIN, 0);
  ledcWrite(IN3_PIN, 0);
  ledcWrite(IN4_PIN, 0);
  smoothedSpeed = 0;
  lastLeftDir = 0;
  lastRightDir = 0;
}

void setHornHardware(bool state) {
  if (HORN_PIN >= 0) {
    digitalWrite(HORN_PIN, state ? HIGH : LOW);
  }
}

// Helper function to set Headlight Output via PWM
void setHeadlightHardware(bool state) {
  actualLightState = state;
  if (state) {
    ledcWrite(HEADLIGHT_PIN, headlightBrightness);
  } else {
    ledcWrite(HEADLIGHT_PIN, 0);
  }
}

        function handleWebSocketMessage(event) {
            if (event.data.startsWith("ALERT_")) {
                showToast(event.data.substring(6));
                return;
            }

            if (event.data.startsWith("LIGHT_STATE_")) {
                let st = event.data.substring(12) === "1";
                updateLightUIState(st);
                return;
            }

            try {
                let data = JSON.parse(event.data);

                if (data.dist !== undefined) document.getElementById('ui-dist').innerText = data.dist + " cm";
                if (data.wifi !== undefined) document.getElementById('wifi-signal-text').innerText = data.wifi + "%";
                if (data.ldr !== undefined) document.getElementById('ui-ldr').innerText = data.ldr;
                if (data.temp !== undefined) {
                    document.getElementById('ui-temp').innerText = data.temp + " °C";
                    updateTemperatureUI(data.temp);
                }
                if (data.blink !== undefined && activeWidget === "FACE") {
                    eyeLeft.classList.toggle('blinking', !!data.blink);
                    eyeRight.classList.toggle('blinking', !!data.blink);
                }
            } catch (e) { }
        }

        function initWebSocket() {
            let wsGateway = `ws://${window.location.hostname}/ws`;
            if (window.location.hostname === "") { wsGateway = "ws://192.168.4.1/ws"; }

            try {
                ws = new WebSocket(wsGateway);
                ws.onmessage = handleWebSocketMessage;

                ws.onopen = function () {
                    wsStatusBadge.innerText = "LIVE";
                    wsStatusBadge.classList.add('ws-connected');
                    sendCmd("CLIENT_CONNECTED");
                };

                ws.onclose = function () {
                    wsStatusBadge.innerText = "OFF";
                    wsStatusBadge.classList.remove('ws-connected');
                    if (wsReconnectTimer) clearTimeout(wsReconnectTimer);
                    wsReconnectTimer = setTimeout(initWebSocket, 2000);
                };

                if (wsHeartbeatTimer) clearInterval(wsHeartbeatTimer);
                wsHeartbeatTimer = setInterval(() => {
                    if (ws && ws.readyState === WebSocket.OPEN) {
                        ws.send("HEARTBEAT");
                    }
                }, 500);

float getDistanceCM() {
  static unsigned long lastMeasure = 0;
  static float cachedDist = 999.0;
  if (millis() - lastMeasure > 60) {
    lastMeasure = millis();
    digitalWrite(TRIG_PIN, LOW);
    delayMicroseconds(2);
    digitalWrite(TRIG_PIN, HIGH);
    delayMicroseconds(10);
    digitalWrite(TRIG_PIN, LOW);
    long duration = pulseIn(ECHO_PIN, HIGH, 6000);
    if (duration == 0) cachedDist = 999.0;
    else cachedDist = duration * 0.034 / 2.0;
  }
  return cachedDist;
}

// ------------------- MOTOR PROTECTION CONTROL -------------------
void applyMotorControl(int targetSpeedVal, int steerVal) {
  float distance = getDistanceCM();

  if (distance <= 7.0 && targetSpeedVal > 0) {
    targetSpeedVal = 0;
    smoothedSpeed = 0;
  }

  if (isParked) {
    targetSpeedVal = 0;
    steerVal = 0;
  }

  // Smooth Speed Ramping (Timing control - millis මත පදනම්ව)
  if (millis() - lastRampTime >= 15) {
    lastRampTime = millis();
    int delta = targetSpeedVal - smoothedSpeed;
    if (abs(delta) <= 2) {
      smoothedSpeed = targetSpeedVal;
    } else if (delta > 0) {
      smoothedSpeed += 2;
    } else {
      smoothedSpeed -= 2;
    }
  }

  int effectiveSpeed = smoothedSpeed;

  if (isParked) {
    stopMotors();
    return;
  }

  // Auto Arrow Light එක ON වී ඇත්නම් සහ Manual Signal දමා නැත්නම් පමණක් Steering අනුව දැල්වේ
  if (autoArrowEnabled && manualSignalState == "OFF" && !isParked) {
    if (steerVal < -15) {
      digitalWrite(ARROW_LEFT_PIN, HIGH);
      digitalWrite(ARROW_RIGHT_PIN,
                   LOW);
    } else if (steerVal > 15) {
      digitalWrite(ARROW_LEFT_PIN, LOW);
      digitalWrite(ARROW_RIGHT_PIN, HIGH);
    } else {
      digitalWrite(ARROW_LEFT_PIN, LOW);
      digitalWrite(ARROW_RIGHT_PIN, LOW);
    }
  }

  // Differential Steering
  int leftMotorSpeed = effectiveSpeed + (steerVal / 2);
  int rightMotorSpeed = effectiveSpeed - (steerVal / 2);

  leftMotorSpeed = constrain(leftMotorSpeed, -100, 100);
  rightMotorSpeed = constrain(rightMotorSpeed, -100, 100);

  // Minimum PWM Mapping (අඩු වේගවලදී Stalling වළක්වා ගැනීමට)
  int leftPwm = 0;
  if (abs(leftMotorSpeed) > 0) {
    leftPwm = map(abs(leftMotorSpeed), 1, 100, MIN_PWM, 255);
  }

  int rightPwm = 0;
  if (abs(rightMotorSpeed) > 0) {
    rightPwm = map(abs(rightMotorSpeed), 1, 100, MIN_PWM, 255);
  }

  // Left Motor Dead-Time Protection (දිශාව මාරු වීමේදී Peak Current ආරක්ෂාව)
  int currentLeftDir = (leftMotorSpeed > 0) ? 1 : ((leftMotorSpeed < 0) ? -1 : 0);
  if (lastLeftDir != 0 && currentLeftDir != 0 && lastLeftDir != currentLeftDir) {
    ledcWrite(IN1_PIN, 0);
    ledcWrite(IN2_PIN, 0);
    delayMicroseconds(500);
  }
  lastLeftDir = currentLeftDir;
  if (leftMotorSpeed > 0) {
    ledcWrite(IN1_PIN, 0);
    ledcWrite(IN2_PIN, leftPwm);
  } else if (leftMotorSpeed < 0) {
    ledcWrite(IN1_PIN, leftPwm);
    ledcWrite(IN2_PIN, 0);
  } else {
    ledcWrite(IN1_PIN, 0);
    ledcWrite(IN2_PIN, 0);
  }  // Right Motor // Dead
  // Dead - Time Protection
  int currentRightDir = (rightMotorSpeed > 0) ? 1 : ((rightMotorSpeed < 0) ? -1 : 0);
  if (lastRightDir != 0 && currentRightDir != 0 && lastRightDir != currentRightDir) {
    ledcWrite(IN3_PIN, 0);
    ledcWrite(IN4_PIN,
              0);
    delayMicroseconds(500);
  }
  lastRightDir = currentRightDir;
  if (rightMotorSpeed > 0) {
    ledcWrite(IN3_PIN, 0);
    ledcWrite(IN4_PIN, rightPwm);
  } else if (rightMotorSpeed < 0) {
    ledcWrite(IN3_PIN, rightPwm);
    ledcWrite(IN4_PIN, 0);
  } else {
    ledcWrite(IN3_PIN, 0);
    ledcWrite(IN4_PIN, 0);
  }
}
void handleAutoMode() {
  float distance = getDistanceCM();
  unsigned long currentMillis = millis();

  switch (autoState) {
    case 0:  // ඉදිරියට ධාවනය
      if (distance < 20.0) {
        applyMotorControl(0, 0);
        autoPreviousMillis = currentMillis;
        autoState = 1;
      } else {
        applyMotorControl(60, 0);
      }
      break;

    case 1:  // නැවතීම
      applyMotorControl(0, 0);
      if (currentMillis - autoPreviousMillis >= 100) {
        autoPreviousMillis = currentMillis;
        autoState = 2;
      }
      break;

    case 2:  // පසුපසට යෑම
      applyMotorControl(-50, 0);
      if (currentMillis - autoPreviousMillis >= 300) {
        autoPreviousMillis = currentMillis;
        autoState = 3;
      }
      break;

    case 3:  // හැරවීම (Turn)
      applyMotorControl(0, -80);
      if (currentMillis - autoPreviousMillis >= 400) {
        autoPreviousMillis = currentMillis;
        autoState = 4;
      }
      break;

    case 4:  // කෙටි විරාමය
      applyMotorControl(0, 0);
      if (currentMillis - autoPreviousMillis >= 100) {
        autoState = 0;
      }
      break;
  }
}

// ------------------- URL DECODING -------------------
int hexValue(char c) {
  if (c >= '0' && c <= '9') return c - '0';
  if (c >= 'A' && c <= 'F') return c - 'A' + 10;
  if (c >= 'a' && c <= 'f') return c - 'a' + 10;
  return -1;
}

String urlDecode(const String &input) {
  String out;
  out.reserve(input.length());
  for (int i = 0; i < input.length(); i++) {
    char c = input[i];
    if (c == '%' && i + 2 < input.length()) {
      int hi = hexValue(input[i + 1]);
      int lo = hexValue(input[i + 2]);
      if (hi >= 0 && lo >= 0) {
        out += char((hi << 4) | lo);
        i += 2;
        continue;
      }
    }
    if (c == '+') out += ' ';
    else out += c;
  }
  return out;
}

// WebSocket Command Processor
void onWsEvent(AsyncWebSocket *server, AsyncWebSocketClient *client,
               AwsEventType type, void *arg, uint8_t *data, size_t len) {

  if (type == WS_EVT_CONNECT) {
    return;
  }

  if (type == WS_EVT_DISCONNECT) {
    if (ws.count() == 0) {
      isConnectedToWeb = false;
      currentSpeed = 0;
      currentSteer = 0;
      sysMode = "RC";
      autoState = 0;
      manualSignalState = "OFF";
      digitalWrite(ARROW_LEFT_PIN, LOW);
      digitalWrite(ARROW_RIGHT_PIN, LOW);
      stopMotors();
      setHornHardware(false);
    }
    return;
  }

  if (type != WS_EVT_DATA) return;

  String msg;
  msg.reserve(len);
  for (size_t i = 0; i < len; i++) msg += (char)data[i];

  if (msg == "CLIENT_CONNECTED") {
    isConnectedToWeb = true;
    lastControlCommandMs = millis();
    ws.textAll("LIGHT_STATE_" + String(actualLightState ? "1" : "0"));
    return;
  }

  if (msg == "HEARTBEAT") {
    lastControlCommandMs = millis();
    return;
  }

  if (msg.startsWith("WIFI_CONNECT|")) {
    int sep = msg.indexOf('|', 13);
    if (sep > 13) {
      String ssid = urlDecode(msg.substring(13, sep));
      String pass = urlDecode(msg.substring(sep + 1));

      currentSpeed = 0;
      currentSteer = 0;
      smoothedSpeed = 0;
      stopMotors();
      digitalWrite(ARROW_LEFT_PIN, LOW);
      digitalWrite(ARROW_RIGHT_PIN, LOW);
      manualSignalState = "OFF";

      WiFi.disconnect(true);
      WiFi.mode(WIFI_STA);
      WiFi.begin(ssid.c_str(), pass.c_str());
      wifiConnectRequested = true;
      wifiConnectStartMs = millis();
    }
    return;
  }

  if (msg.startsWith("SPEED_")) {
    currentSpeed = constrain(msg.substring(6).toInt(), -100, 100);
    lastControlCommandMs = millis();
    return;
  }

  if (msg.startsWith("STEER_")) {
    currentSteer = constrain(msg.substring(6).toInt(), -100, 100);
    lastControlCommandMs = millis();
    return;
  }

  if (msg == "PARK_ON") {
    isParked = true;
    currentSpeed = 0;
    currentSteer = 0;
    manualSignalState = "OFF";
    stopMotors();
    return;
  }

  if (msg == "PARK_OFF") {
    isParked = false;
    currentSpeed = 0;
    currentSteer = 0;
    smoothedSpeed = 0;
    digitalWrite(ARROW_LEFT_PIN, LOW);
    digitalWrite(ARROW_RIGHT_PIN, LOW);
    return;
  }

  if (msg == "TOGGLE_AUTO_ARROW") {
    autoArrowEnabled = !autoArrowEnabled;
    if (!autoArrowEnabled) {
      digitalWrite(ARROW_LEFT_PIN, LOW);
      digitalWrite(ARROW_RIGHT_PIN, LOW);
    }
    ws.textAll(autoArrowEnabled ? "ALERT_Auto Arrow Light: ON" : "ALERT_Auto Arrow Light: OFF");
    return;
  }

  if (msg == "SIGNAL_L") {
    manualSignalState = "L";
    signalBlinkState = false;
    lastSignalBlink = millis();
    digitalWrite(ARROW_RIGHT_PIN, LOW);
    return;
  }

  if (msg == "SIGNAL_R") {
    manualSignalState = "R";
    signalBlinkState = false;
    lastSignalBlink = millis();
    digitalWrite(ARROW_LEFT_PIN, LOW);
    return;
  }

  if (msg == "SIGNAL_OFF") {
    manualSignalState = "OFF";
    signalBlinkState = false;
    digitalWrite(ARROW_LEFT_PIN, LOW);
    digitalWrite(ARROW_RIGHT_PIN, LOW);
    return;
  }

  if (msg == "MODE_RC") {
    sysMode = "RC";
    autoState = 0;
    currentSpeed = 0;
    currentSteer = 0;
    smoothedSpeed = 0;
    stopMotors();
    return;
  }

  if (msg == "MODE_AUTO") {
    sysMode = "AUTO";
    autoState = 0;
    autoPreviousMillis = millis();
    currentSpeed = 0;
    currentSteer = 0;
    smoothedSpeed = 0;
    stopMotors();
    return;
  }

  if (msg == "HORN_ON") {
    setHornHardware(true);
    return;
  }

  if (msg == "HORN_OFF") {
    setHornHardware(false);
    return;
  }

  if (msg == "LIGHT_ON") {
    manualLightState = true;
    setHeadlightHardware(true);
    ws.textAll("LIGHT_STATE_1");
    return;
  }

  if (msg == "LIGHT_OFF") {
    manualLightState = false;
    setHeadlightHardware(false);
    ws.textAll("LIGHT_STATE_0");
    return;
  }

  if (msg.startsWith("HEADLIGHT_BRIGHTNESS_")) {
    int val = constrain(msg.substring(21).toInt(), 0, 100);
    headlightBrightness = map(val, 0, 100, 0, 255);
    if (actualLightState) setHeadlightHardware(true);
    return;
  }

  if (msg == "TOGGLE_AUTO_LDR") {
    autoLdrEnabled = !autoLdrEnabled;
    if (!autoLdrEnabled && !manualLightState) {
      setHeadlightHardware(false);
      ws.textAll("LIGHT_STATE_0");
    }
    ws.textAll(autoLdrEnabled ? "ALERT_Auto LDR Control: ON" : "ALERT_Auto LDR Control: OFF");
    return;
  }

  if (msg == "RGB_ON") {
    isRgbOn = true;
    applyRgbHardware();
    return;
  }

  if (msg == "RGB_OFF") {
    isRgbOn = false;
    applyRgbHardware();
    return;
  }

  if (msg.startsWith("RGB_COLOR_")) {
    String hexColor = msg.substring(10);
    if (hexColor.length() == 6) {
      long number = strtol(hexColor.c_str(), NULL, 16);
      lastRgbR = (number >> 16) & 0xFF;
      lastRgbG = (number >> 8) & 0xFF;
      lastRgbB = number & 0xFF;
      isRgbOn = true;
      applyRgbHardware();
    }
    return;
  }

  if (msg.startsWith("WIDGET_")) {
    activeWidget = msg.substring(7);
    return;
  }

  if (msg.startsWith("CUSTOM_TEXT_")) {
    oledCustomText = msg.substring(12);
    activeWidget = "TEXT";
    return;
  }

  if (msg.startsWith("DATETIME_")) {
    int idx = msg.indexOf('_', 9);
    if (idx > 9) {
      liveTimeStr = msg.substring(9, idx);
      liveDateStr = msg.substring(idx + 1);
    }
    return;
  }
}

// ------------------- SETUP -------------------
void setup() {
  Serial.begin(115200);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(LDR_PIN, INPUT);
  pinMode(ARROW_LEFT_PIN, OUTPUT);
  pinMode(ARROW_RIGHT_PIN, OUTPUT);
  pinMode(RED_PIN, OUTPUT);
  pinMode(GREEN_PIN, OUTPUT);
  pinMode(BLUE_PIN, OUTPUT);
  if (HORN_PIN >= 0) {
    pinMode(HORN_PIN, OUTPUT);
    digitalWrite(HORN_PIN, LOW);
  }

  // Attach PWM for Motors
  ledcAttach(IN1_PIN, PWM_FREQ, PWM_RES);
  ledcAttach(IN2_PIN, PWM_FREQ, PWM_RES);
  ledcAttach(IN3_PIN, PWM_FREQ, PWM_RES);
  ledcAttach(IN4_PIN, PWM_FREQ, PWM_RES);

  // Attach PWM for Headlight Brightness
  ledcAttach(HEADLIGHT_PIN, PWM_FREQ, PWM_RES);
  setHeadlightHardware(false);

  Wire.begin();
  Wire.setClock(400000);
  u8g2.begin();

  WiFi.mode(WIFI_AP);
  WiFi.softAP("Smart_RC_Car", "12345678");
  currentIP = WiFi.softAPIP().toString();

  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request) {
    if (isAPMode) {
      request->send_P(200, "text/html", WIFI_SETUP_HTML);
    } else {
      request->send_P(200, "text/html", REMOTE_HTML);
    }
  });

  ws.onEvent(onWsEvent);
  server.addHandler(&ws);
  server.begin();
}

// ------------------- MAIN LOOP -------------------
unsigned long lastUpdate = 0;
unsigned long lastOledRefresh = 0;

void loop() {
  ws.cleanupClients();

  if (wifiConnectRequested) {
    if (WiFi.status() == WL_CONNECTED) {
      isAPMode = false;
      currentIP = WiFi.localIP().toString();
      wifiConnectRequested = false;
      isConnectedToWeb = false;
    } else if (millis() - wifiConnectStartMs > 15000) {
      WiFi.disconnect(true);
      WiFi.mode(WIFI_AP);
      WiFi.softAP("Smart_RC_Car", "12345678");
      currentIP = WiFi.softAPIP().toString();
      isAPMode = true;
      wifiConnectRequested = false;
      isConnectedToWeb = false;
      currentSpeed = 0;
      currentSteer = 0;
      sysMode = "RC";
      autoState = 0;
      stopMotors();
      setHornHardware(false);
    }
  }

  // ------------------- COMMUNICATION FAILSAFE -------------------
  if (isConnectedToWeb && millis() - lastControlCommandMs > CONTROL_TIMEOUT_MS) {
    currentSpeed = 0;
    currentSteer = 0;
    smoothedSpeed = 0;
    stopMotors();
  }

  static unsigned long lastParkBlink = 0;
  static bool parkBlinkState = false;

  if (isParked) {
    if (millis() - lastParkBlink > 500) {
      lastParkBlink = millis();
      parkBlinkState = !parkBlinkState;
      digitalWrite(ARROW_LEFT_PIN, parkBlinkState ? HIGH : LOW);
      digitalWrite(ARROW_RIGHT_PIN, parkBlinkState ? HIGH : LOW);
    }
  } else if (manualSignalState != "OFF") {
    if (millis() - lastSignalBlink >= 400) {
      lastSignalBlink = millis();
      signalBlinkState = !signalBlinkState;

      if (manualSignalState == "L") {
        digitalWrite(ARROW_LEFT_PIN, signalBlinkState ? HIGH : LOW);
        digitalWrite(ARROW_RIGHT_PIN, LOW);
      } else if (manualSignalState == "R") {
        digitalWrite(ARROW_RIGHT_PIN, signalBlinkState ? HIGH : LOW);
        digitalWrite(ARROW_LEFT_PIN, LOW);
      }
    }
  } else if (!autoArrowEnabled) {
    digitalWrite(ARROW_LEFT_PIN, LOW);
    digitalWrite(ARROW_RIGHT_PIN, LOW);
  }

  currentLdrValue = analogRead(LDR_PIN);

  if (autoLdrEnabled) {
    if (currentLdrValue < 1200) {
      if (!actualLightState) {
        setHeadlightHardware(true);
        ws.textAll("LIGHT_STATE_1");
      }
    } else if (currentLdrValue > 1400) {
      if (!manualLightState && actualLightState) {
        setHeadlightHardware(false);
        ws.textAll("LIGHT_STATE_0");
      }
    }
  }

  if (millis() - lastBlinkTime > (isBlinking ? 150 : 3500)) {
    lastBlinkTime = millis();
    isBlinking = !isBlinking;

    if (activeWidget == "FACE") {
      String blinkJson = "{\"blink\":" + String(isBlinking ? "true" : "false") + "}";
      ws.textAll(blinkJson);
    }
  }

  if (sysMode == "RC") {
    applyMotorControl(currentSpeed, currentSteer);
  } else if (sysMode == "AUTO") {
    handleAutoMode();
  }

  static unsigned long lastWifiCheck = 0;
  if (millis() - lastWifiCheck > 1000) {
    lastWifiCheck = millis();
    int quality = 0;

    if (WiFi.status() == WL_CONNECTED) {
      int rssi = WiFi.RSSI();
      if (rssi <= -100) quality = 0;
      else if (rssi >= -50) quality = 100;
      else quality = 2 * (rssi + 100);
    } else if (isAPMode) {
      quality = 100;
    }

    String wifiJson = "{\"wifi\":" + String(quality) + "}";
    ws.textAll(wifiJson);
  }

  if (millis() - lastOledRefresh > 120) {
    lastOledRefresh = millis();
    drawOLED();
  }

  if (millis() - lastUpdate > 300) {
    lastUpdate = millis();
    int envTemp = getInternalTemp();
    float dist = getDistanceCM();

    String jsonMsg = "{\"temp\":" + String(envTemp) + ",\"dist\":" + String(dist, 1) + ",\"ldr\":" + String(currentLdrValue) + "}";
    ws.textAll(jsonMsg);
  }
}
